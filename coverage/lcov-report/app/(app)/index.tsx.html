function getStatusLabel(status: PickupOrder["status"]) {
	switch (status) {
		case "confirmed":
			return "Đã xác nhận";
		case "on_the_way":
			return "Đang tới";
		default:
			return "Đang xử lý";
	}
}

function formatDate(value: string) {
	return new Intl.DateTimeFormat("vi-VN", {
		day: "2-digit",
		month: "2-digit",
		year: "numeric",
	}).format(new Date(`${value}T00:00:00`));
}
